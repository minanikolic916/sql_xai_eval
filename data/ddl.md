SCHEMA 1:

CREATE TABLE WORKER
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(30),
SURNAME VARCHAR2(50),
ROLE VARCHAR2(30),
TYPE_OF_EMPLOYMENT VARCHAR2(20)
);

CREATE TABLE SKIES
(
ID NUMBER PRIMARY KEY,
MANUFACTURER VARCHAR2(30),
SKI_MODEL VARCHAR2(50),
LENGTH NUMBER,
CATEGORY VARCHAR2(20)
);

CREATE TABLE SKIER
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(30),
SURNAME VARCHAR2(50),
GENDER CHAR(1),
AGE NUMBER,
SKIER_LEVEL VARCHAR2(20)
);

CREATE TABLE SKI_RENT
(
ID NUMBER PRIMARY KEY,
WORKER_ID NUMBER REFERENCES WORKER(ID),
SKI_ID NUMBER REFERENCES SKIES(ID),
SKIER_ID NUMBER REFERENCES SKIER(ID),
PRICE NUMBER,
RENT_DATE DATE,
RENT_DURATION NUMBER,
PRICE_REDUCTION VARCHAR2(5)
)

SCHEMA 2:

CREATE TABLE REALTOR
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(30),
SURNAME VARCHAR2(50),
SALARY NUMBER,
YEARS_OF_SERVICE NUMBER
);

CREATE TABLE PROPERTY
(
ID NUMBER PRIMARY KEY,
LISTING_NAME VARCHAR2(30),
LOCATION VARCHAR2(50),
CONSTRUCTION_YEAR NUMBER,
STRUCTURE VARCHAR2(20),
PARKING VARCHAR2(5)
);

CREATE TABLE TENANT
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(30),
SURNAME VARCHAR2(50),
GENDER CHAR(1),
AGE NUMBER
);

CREATE TABLE CONTRACT
(
ID NUMBER PRIMARY KEY,
REALTOR_ID NUMBER REFERENCES REALTOR(ID),
PROPERTY_ID NUMBER REFERENCES PROPERTY(ID),
TENANT_ID NUMBER REFERENCES TENANT(ID),
RENTAL_DATE DATE,
PRICE NUMBER,
CONTRACT_LENGTH NUMBER
);

SCHEMA 3:

CREATE TABLE VETERINARIAN
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(30),
SURNAME VARCHAR2(30),
SPECIALIZATION VARCHAR2(10),
YEARS_OF_EXPERIENCE NUMBER,
SALARY NUMBER
);

CREATE TABLE PET
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(30),
PET_TYPE VARCHAR2(20),
ALLERGIES VARCHAR2(100)
);

CREATE TABLE OWNER
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(30),
SURNAME VARCHAR2(50),
CITY VARCHAR2(20),
PHONE VARCHAR2(20)
);

CREATE TABLE EXAM
(
ID NUMBER PRIMARY KEY,
VETERINARIAN_ID NUMBER REFERENCES VETERINARIAN(ID),
PET_ID NUMBER REFERENCES PET(ID),
OWNER_ID NUMBER REFERENCES OWNER(ID),
EXAM_DATE DATE,
TYPE_OF_EXAM VARCHAR2(20),
DURATION NUMBER,
PRICE NUMBER,
THERAPY_PRESCRIBED VARCHAR2(5)
);

SCHEMA 4:

CREATE TABLE COACH
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(20),
SURNAME VARCHAR2(50),
SPECIALIZATION VARCHAR2(20),
NUMBER_OF_ACTIVE_MEMBERS NUMBER,
PROVISION_PERCENTAGE NUMBER
);

CREATE TABLE PACKAGE
(
ID NUMBER PRIMARY KEY,
PACKAGE_NAME VARCHAR2(20),
DURATION_MONTHS NUMBER,
PRICE NUMBER,
DISCOUNT VARCHAR2(5),
TARGET_GROUP VARCHAR2(10)
);

CREATE TABLE GYM_MEMBER
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(20),
SURNAME VARCHAR2(50),
AGE NUMBER,
HEIGHT NUMBER,
WEIGHT NUMBER,
STATUS VARCHAR2(10)
);

CREATE TABLE WORKOUT
(
ID NUMBER PRIMARY KEY,
COACH_ID NUMBER REFERENCES COACH(ID),
PACKAGE_ID NUMBER REFERENCES PACKAGE(ID),
GYM_MEMBER_ID NUMBER REFERENCES GYM_MEMBER(ID),
WORKOUT_DATE DATE,
PLACE VARCHAR2(20),
DURATION NUMBER,
GRADE NUMBER
);

SCHEMA 5:

CREATE TABLE WAITER
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(20),
SURNAME VARCHAR2(50),
SALARY NUMBER
);

CREATE TABLE MEAL
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(20),
MEAL_TYPE VARCHAR2(10),
NUMBER_OF_CALORIES NUMBER,
GLUTEN VARCHAR2(5)
);

CREATE TABLE GUEST
(
ID NUMBER PRIMARY KEY,
NAME VARCHAR2(20),
SURNAME VARCHAR2(50),
CITY VARCHAR2(20),
PHONE VARCHAR2(11),
REGULAR_CUSTOMERS VARCHAR2(5)
);

CREATE TABLE ORDERS
(
ID NUMBER PRIMARY KEY,
WAITER_ID NUMBER REFERENCES WAITER(ID),
MEAL_ID NUMBER REFERENCES MEAL(ID),
GUEST_ID NUMBER REFERENCES GUEST(ID),
ORDER_DATE DATE,
PRICE NUMBER,
PAYMENT_METHOD VARCHAR2(10),
TIP VARCHAR2(5)
);
